def consent_guard(request):
    """
    TrustMesh ConsentGuard: Evaluates Partner Identity,
    Access Token, Customer Consent, Scopes, and Rate Limits.
    """
    # Step 1: Authenticate the partner
    if not authenticate_partner(request.partner_token):
        return deny("INVALID_PARTNER_AUTHENTICATION")

    # Step 2: Validate the access token
    if not validate_token(request.access_token):
        return deny("INVALID_OR_EXPIRED_TOKEN")

    # Step 3: Check customer consent
    consent = get_customer_consent(
        request.customer_id,
        request.partner_id
    )

    if consent is None or not consent.is_active:
        return deny("CONSENT_MISSING_OR_REVOKED")

    # Step 4: Check requested permission and data scope
    if not request.scope.issubset(consent.allowed_scopes):
        return deny("INSUFFICIENT_PERMISSION")

    # Step 5: Check purpose and consent expiry
    if consent.is_expired or not purpose_is_allowed(
        request.purpose, consent
    ):
        return deny("CONSENT_EXPIRED_OR_PURPOSE_MISMATCH")

    # Step 6: Apply API security policies & rate limits
    if not rate_limit_ok(request.partner_id):
        return deny("RATE_LIMIT_EXCEEDED")

    # Step 7: Forward only the permitted request
    response = call_banking_api(request)
    return allow(response)
          

        


        
        

          

            
Reason Codes & Security Enforcements

            Zero Information Leakage
          


          

            

              

                INVALID_PARTNER_AUTHENTICATION
                HTTP 401
              

              

                Triggered if mTLS identity certificate or partner API token fails cryptographic validation. Prevents unauthorized third parties from probing banking endpoints.
              

            


            

              

                CONSENT_MISSING_OR_REVOKED
                HTTP 403
              

              

                Enforces the principle of customer data sovereignty. If a customer has revoked permission in their banking app, the request is immediately terminated before reaching Core Banking.
              

            


            

              

                INSUFFICIENT_PERMISSION
                HTTP 403
              

              

                Account-boundary and operation-boundary enforcement. If customer granted access only to Account A balance, an inquiry for Account B or transactions is strictly blocked.
              

            


            

              

                CONSENT_EXPIRED_OR_PURPOSE_MISMATCH
                HTTP 403
              

              

                Consent agreements must have strict validity windows (e.g. 90 days) and purpose binding (e.g. LOAN_UNDERWRITING vs. unapproved marketing usage).
              

            


            

              

                RATE_LIMIT_EXCEEDED
                HTTP 429
              

              

                Prevents data scraping or Denial of Service attacks on core banking ledgers through token-bucket rate limiting per partner tier.
